local function checkTarget(obj, myPos, currentMinDist, currentClosest)
    if obj:IsA("Model") and obj ~= localPlayer.Character then
        local hum = obj:FindFirstChildOfClass("Humanoid")
        local hrp = obj:FindFirstChild("HumanoidRootPart")
        if hum and hrp and hum.Health > 0 then
            local dist = (hrp.Position - myPos).Magnitude
            if dist < currentMinDist then return dist, hrp end
        end
    end
    return currentMinDist, currentClosest
end

local function getNearestTarget()
    local myChar = localPlayer.Character
    if not myChar or not myChar:FindFirstChild("HumanoidRootPart") then return nil end
    local myPos = myChar.HumanoidRootPart.Position
    local closestHRP = nil
    local minDist = math.huge
    local liveFolder = workspace:FindFirstChild("Live")
    
    if liveFolder then
        for _, target in pairs(liveFolder:GetChildren()) do
            if target ~= myChar then
                minDist, closestHRP = checkTarget(target, myPos, minDist, closestHRP)
            end
        end
    else
        for _, targetPlayer in pairs(Players:GetPlayers()) do
            if targetPlayer ~= localPlayer and targetPlayer.Character then
                minDist, closestHRP = checkTarget(targetPlayer.Character, myPos, minDist, closestHRP)
            end
        end
    end
    return closestHRP
end

local function isVisible(targetPart)
    local origin = camera.CFrame.Position
    local direction = (targetPart.Position - origin).Unit * (targetPart.Position - origin).Magnitude
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = {localPlayer.Character, camera}
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.IgnoreWater = true

    local result = workspace:Raycast(origin, direction, params)
    if result then
        if result.Instance:IsDescendantOf(targetPart.Parent) then return true end
        return false
    end
    return true
end

local MAX_DISTANCE = 1500 -- Maksimum mesafe sınırı (stud/metre)

local function getAimbotTarget()
    local mouseLocation = UserInputService:GetMouseLocation()
    local closestTarget = nil
    local shortestDistance = aimbotFOV

    local localCharacter = localPlayer.Character
    local localRoot = localCharacter and (localCharacter:FindFirstChild("HumanoidRootPart") or localCharacter:FindFirstChild("Head"))
    
    if not localRoot then return nil end

    for _, v in pairs(Players:GetPlayers()) do
        if v ~= localPlayer and v.Character and v.Character:FindFirstChild("Head") and v.Character:FindFirstChild("Humanoid") then
            if v.Character.Humanoid.Health > 0 then
                if localPlayer.Team == nil or v.Team == nil or localPlayer.Team ~= v.Team then
                    local worldDistance = (v.Character.Head.Position - localRoot.Position).Magnitude
                    if worldDistance <= MAX_DISTANCE then
                        local targetPos, onScreen = camera:WorldToViewportPoint(v.Character.Head.Position)
                        if onScreen and targetPos.Z > 0 then
                            local distance = (Vector2.new(targetPos.X, targetPos.Y) - mouseLocation).Magnitude
                            if distance < shortestDistance then
                                if isVisible(v.Character.Head) then
                                    closestTarget = v.Character.Head
                                    shortestDistance = distance
                                end
                            end
                        end
                    end
                end
            end
        end
    end
    return closestTarget
end

local function makeDraggable(gui)
    local dragging, dragInput, dragStart, startPos
    gui.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = gui.Position
        end
    end)
    gui.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            gui.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
    gui.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end

local sliderValues = {} 
local function createSlider(text, parent, min, max, default, callback)
    local container = Instance.new("Frame", parent)
    container.Size = UDim2.new(0.9, 0, 0, 45); container.BackgroundTransparency = 1
    container.ZIndex = 9010
    
    local label = Instance.new("TextLabel", container)
    label.Size = UDim2.new(1, 0, 0, 15); label.BackgroundTransparency = 1
    label.Text = text .. ": " .. default; label.TextColor3 = Color3.new(1,1,1)
    label.Font = Enum.Font.GothamBold; label.TextSize = 12; label.TextXAlignment = Enum.TextXAlignment.Left
    label.ZIndex = 9015
    
    local bg = Instance.new("Frame", container)
    bg.Size = UDim2.new(1, 0, 0, 12); bg.Position = UDim2.new(0, 0, 0, 22)
    bg.BackgroundColor3 = Color3.fromRGB(30, 30, 35)
    bg.ZIndex = 9015
    Instance.new("UICorner", bg).CornerRadius = UDim.new(1,0)
    
    local fill = Instance.new("Frame", bg)
    fill.Size = UDim2.new((default-min)/(max-min), 0, 1, 0)
    fill.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
    fill.ZIndex = 9016
    Instance.new("UICorner", fill).CornerRadius = UDim.new(1,0)
    
    sliderValues[text] = {label = label, fill = fill, min = min, max = max}
    
    local isDragging = false
    bg.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then isDragging = true end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then isDragging = false end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if isDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local mousePos = UserInputService:GetMouseLocation().X
            local bgPos = bg.AbsolutePosition.X; local bgSize = bg.AbsoluteSize.X
            local pct = math.clamp((mousePos - bgPos) / bgSize, 0, 1)
            fill.Size = UDim2.new(pct, 0, 1, 0)
            local val = math.floor(min + (max - min) * pct)
            label.Text = text .. ": " .. val
            callback(val)
        end
    end)
end

local function updateSliderUI(text, newVal)
    if sliderValues[text] then
        local data = sliderValues[text]
        local pct = math.clamp((newVal - data.min) / (data.max - data.min), 0, 1)
        data.fill.Size = UDim2.new(pct, 0, 1, 0)
        data.label.Text = text .. ": " .. newVal
    end
end

local function startEscape()
    if isEscaping then return end
    local char = localPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    
    if char and hrp and hum then
        isEscaping = true
        local originalCFrame = hrp.CFrame
        local destroyHeight = game.Workspace.FallenPartsDestroyHeight
        local escapePos = Vector3.new(0, destroyHeight + 2.5, 0)
        local escapeRot = CFrame.Angles(math.rad(90), 0, 0)
        hum.PlatformStand = true; hum.AutoRotate = false
        
        local bp = Instance.new("BodyPosition", hrp); bp.MaxForce = Vector3.new(1, 1, 1) * math.huge; bp.Position = escapePos; bp.P = 50000 
        local bg = Instance.new("BodyGyro", hrp); bg.MaxTorque = Vector3.new(1, 1, 1) * math.huge; bg.CFrame = escapeRot; bg.P = 500000; bg.D = 500    
        hrp.CFrame = CFrame.new(escapePos) * escapeRot
        escapeBtn.Text = "Kaçılıyor (3s)..."
        
        task.wait(3)
        if char and hrp and hum then
            bp:Destroy(); bg:Destroy()
            hum.PlatformStand = false; hum.AutoRotate = true
            hrp.AssemblyLinearVelocity = Vector3.new(0,0,0); hrp.AssemblyAngularVelocity = Vector3.new(0,0,0)
            hrp.CFrame = originalCFrame
            task.wait(0.1)
            workspace.CurrentCamera.CameraType = Enum.CameraType.Custom; workspace.CurrentCamera.CameraSubject = hum
        end
        escapeBtn.Text = "det kontur yememe"
        isEscaping = false
    end
end

                local att = Instance.new("Attachment", hrp); att.Name = "CustomFlyAttachment"
                flyBodyVelocity = Instance.new("LinearVelocity")
                flyBodyVelocity.Name = "CustomFlyVelocity" ; flyBodyVelocity.MaxForce = 10^10 ; flyBodyVelocity.VelocityConstraintMode = Enum.VelocityConstraintMode.Vector; flyBodyVelocity.Attachment0 = att; flyBodyVelocity.Parent = hrp

                flyBodyGyro = Instance.new("AlignOrientation")
                flyBodyGyro.Name = "CustomFlyGyro" ; flyBodyGyro.MaxTorque = 10^10; flyBodyGyro.Responsiveness = 200; flyBodyGyro.Mode = Enum.OrientationAlignmentMode.OneAttachment; flyBodyGyro.Attachment0 = att; flyBodyGyro.Parent = hrp
                humanoid.PlatformStand = true
            else
                if hrp:FindFirstChild("CustomFlyVelocity") then hrp.CustomFlyVelocity:Destroy() end
                if hrp:FindFirstChild("CustomFlyGyro") then hrp.CustomFlyGyro:Destroy() end
                if hrp:FindFirstChild("CustomFlyAttachment") then hrp.CustomFlyAttachment:Destroy() end
                flyBodyVelocity = nil; flyBodyGyro = nil
                humanoid.PlatformStand = false
                humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
            end
        end
    end

    if input.KeyCode == Enum.KeyCode.R and rKeyEnabled and not isAttached then
        local targetHRP = getNearestTarget()
        if targetHRP then
            isAttached = true
            local character = localPlayer.Character
            local myHRP = character.HumanoidRootPart
            local myHumanoid = character:FindFirstChildOfClass("Humanoid")
            local startT = tick()
            local conn
            
            conn = RunService.Heartbeat:Connect(function()
                if tick() - startT > 0.5 or not targetHRP.Parent or not myHRP.Parent then
                    isAttached = false; myHRP.AssemblyLinearVelocity = Vector3.zero; myHRP.AssemblyAngularVelocity = Vector3.zero 
                    if myHumanoid then myHumanoid:ChangeState(Enum.HumanoidStateType.Landed) end
                    conn:Disconnect() 
                    return
                end
                myHRP.CFrame = CFrame.new(targetHRP.Position + Vector3.new(0, 3, 0)) * CFrame.Angles(math.rad(90), 0, 0)
                myHRP.AssemblyLinearVelocity = Vector3.new(0, 200, 0)
            end)
        end
    end

    if input.KeyCode == Enum.KeyCode.E and yonelmeEnabled then
        isYonelmeActive = not isYonelmeActive
        local myChar = localPlayer.Character
        local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")
        local humanoid = myChar and myChar:FindFirstChild("Humanoid")

        if not isYonelmeActive then
            if yonelmeConn then yonelmeConn:Disconnect() end
            if currentAlignOri then currentAlignOri:Destroy() end
            if currentAtt then currentAtt:Destroy() end
            if humanoid then humanoid.AutoRotate = true end
            if myHrp then myHrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0); myHrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0) end
            return 
        end

        local targetHRP = getNearestTarget()
        if targetHRP and myHrp and humanoid then
            if yonelmeConn then yonelmeConn:Disconnect() end
            currentAtt = Instance.new("Attachment", myHrp)
            currentAlignOri = Instance.new("AlignOrientation")
            currentAlignOri.Mode = Enum.OrientationAlignmentMode.OneAttachment; currentAlignOri.Attachment0 = currentAtt; currentAlignOri.MaxTorque = 10^7; currentAlignOri.Responsiveness = 200; currentAlignOri.CFrame = CFrame.Angles(math.rad(90), 0, 0); currentAlignOri.Parent = myHrp
            humanoid.AutoRotate = false
            
            yonelmeConn = RunService.Heartbeat:Connect(function()
                if not targetHRP.Parent or not myHrp.Parent then
                    isYonelmeActive = false
                    if yonelmeConn then yonelmeConn:Disconnect() end
                    if currentAlignOri then currentAlignOri:Destroy() end
                    if currentAtt then currentAtt:Destroy() end
                    humanoid.AutoRotate = true
                    return
                end
                local currentY = myHrp.Position.Y 
                local targetPos = targetHRP.Position
                myHrp.CFrame = CFrame.new(targetPos.X, currentY, targetPos.Z) * CFrame.Angles(math.rad(90), 0, 0)
            end)
        else
            isYonelmeActive = false
        end
    end
end)

    local char = localPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    
    if showFovEnabled then
        local mouseLocation = UserInputService:GetMouseLocation()
        fovFrame.Position = UDim2.new(0, mouseLocation.X, 0, mouseLocation.Y)

        local target = getAimbotTarget()

            if onScreen then
                local mouseLocation = UserInputService:GetMouseLocation()
                local moveX = (targetPos.X - mouseLocation.X)
                local moveY = (targetPos.Y - mouseLocation.Y)
                if mousemoverel then
                    mousemoverel(moveX * aimbotSmoothness, moveY * aimbotSmoothness)
                end
            end
        end
    end

    if hum then
        if speedLoopEnabled then hum.WalkSpeed = selectedSpeed end
        if jumpLoopEnabled then hum.UseJumpPower = true; hum.JumpPower = selectedJump end
    end

    for _, p in pairs(Players:GetPlayers()) do
        if p ~= localPlayer and p.Character then
            local h = p.Character:FindFirstChild("AydoESP") or Instance.new("Highlight", p.Character)
            h.Name = "AydoESP"; h.Enabled = espEnabled
        end
    end
    
    if monsterEspEnabled then
        for _, obj in pairs(workspace:GetChildren()) do
            if obj:IsA("Model") and obj:FindFirstChildOfClass("Humanoid") and not Players:GetPlayerFromCharacter(obj) then
                local h = obj:FindFirstChild("MonsterESP") or Instance.new("Highlight", obj)
                h.Name = "MonsterESP"; h.Enabled = true; h.FillColor = Color3.new(1, 0.5, 0)
            end
        end
    end

    if isFlying and flyBodyVelocity and flyBodyGyro then
        local dir = Vector3.zero
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0, 1, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then dir = dir - Vector3.new(0, 1, 0) end
        flyBodyVelocity.VectorVelocity = dir.Magnitude > 0 and dir.Unit * flySpeed or Vector3.zero
        flyBodyGyro.CFrame = camera.CFrame
    end

    if noclipEnabled and char then
        for _, v in pairs(char:GetDescendants()) do if v:IsA("BasePart") then v.CanCollide = false end end
    end
    
    if ragdollEnabled and hum then 
        for _, t in pairs(hum:GetPlayingAnimationTracks()) do t:Stop() end
    end
end)

RunService.Heartbeat:Connect(function(delta)
    if tpwalkEnabled then
        local char = localPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if char and hum and hrp and hum.MoveDirection.Magnitude > 0 then
            hrp.CFrame = hrp.CFrame + (hum.MoveDirection * (tpwalkSpeed * delta * 30))
        end
    end

    if not autoBlockEnabled then 
        if isHoldingF then
            VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.F, false, game)
            isHoldingF = false
        end
        return 
    end

    local myChar = localPlayer.Character
    local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myHrp then return end
    local targetHrp = getNearestTarget()
    
    if targetHrp then
        local targetChar = targetHrp.Parent
        local distance = (targetHrp.Position - myHrp.Position).Magnitude
        if distance <= 15 then
            local hasHunterFists = targetChar:FindFirstChild("HunterFists") and targetChar.HunterFists:IsA("Accessory")
            local hasM1ing = targetChar:FindFirstChild("M1ing") and targetChar.M1ing:IsA("Accessory")
            if hasHunterFists or hasM1ing then
                local targetPos = Vector3.new(targetHrp.Position.X, myHrp.Position.Y, targetHrp.Position.Z)
                myHrp.CFrame = CFrame.lookAt(myHrp.Position, targetPos)
                if not isHoldingF then
                    VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.F, false, game)
                    isHoldingF = true
                end
            else
                if isHoldingF then VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.F, false, game); isHoldingF = false end
            end
        else
            if isHoldingF then VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.F, false, game); isHoldingF = false end
        end
    else
        if isHoldingF then VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.F, false, game); isHoldingF = false end
    end
end)
