local function checkTarget(obj, myPos, currentMinDist, currentClosest)
    -- Bu yardımcı fonksiyon bir objenin geçerli bir hedef olup olmadığını kontrol eder
    if obj:IsA("Model") and obj ~= localPlayer.Character then
        local hum = obj:FindFirstChildOfClass("Humanoid")
        local hrp = obj:FindFirstChild("HumanoidRootPart")

        if hum and hrp and hum.Health > 0 then
            local dist = (hrp.Position - myPos).Magnitude
            if dist < currentMinDist then
                return dist, hrp
            end
        end
    end
    return currentMinDist, currentClosest
end

local function getNearestTarget()
    local myChar = localPlayer.Character
    if not myChar or not myChar:FindFirstChild("HumanoidRootPart") then return nil end
    
    local myPos = myChar.HumanoidRootPart.Position
    local closestHRP = nil
    local minDist = math.huge

    -- 1. ADIM: Oyuncuları direkt servis üzerinden al (Workspace'te nerede olurlarsa olsunlar bulur)
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= localPlayer and player.Character then
            minDist, closestHRP = checkTarget(player.Character, myPos, minDist, closestHRP)
        end
    end

    -- 2. ADIM: Workspace'in 1. ve 2. katmanındaki Dummy'leri tara
    for _, child in pairs(workspace:GetChildren()) do
        -- Önce 1. katmana bak (Dummy direkt Workspace'teyse)
        minDist, closestHRP = checkTarget(child, myPos, minDist, closestHRP)

        -- Eğer bu bir Klasör veya Model ise, içine girip 2. katmana bak
        if child:IsA("Folder") or (child:IsA("Model") and child ~= myChar and not Players:GetPlayerFromCharacter(child)) then
            for _, subChild in pairs(child:GetChildren()) do
                minDist, closestHRP = checkTarget(subChild, myPos, minDist, closestHRP)
            end
        end
    end

    return closestHRP
end

local function createSlider(text, parent, min, max, default, callback)
    local container = Instance.new("Frame", parent)
    container.Size = UDim2.new(0.9, 0, 0, 40); container.BackgroundTransparency = 1
    
    local label = Instance.new("TextLabel", container)
    label.Size = UDim2.new(1, 0, 0, 15); label.BackgroundTransparency = 1
    label.Text = text .. ": " .. default; label.TextColor3 = Color3.new(1,1,1)
    label.Font = Enum.Font.GothamBold; label.TextSize = 12; label.TextXAlignment = Enum.TextXAlignment.Left
    
    local bg = Instance.new("Frame", container)
    bg.Size = UDim2.new(1, 0, 0, 10); bg.Position = UDim2.new(0, 0, 0, 20)
    bg.BackgroundColor3 = Color3.fromRGB(30, 30, 35)
    Instance.new("UICorner", bg).CornerRadius = UDim.new(1,0)
    local stroke = Instance.new("UIStroke", bg); stroke.Color = Color3.fromRGB(60, 60, 70)
    
    local fill = Instance.new("Frame", bg)
    fill.Size = UDim2.new((default-min)/(max-min), 0, 1, 0)
    fill.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
    Instance.new("UICorner", fill).CornerRadius = UDim.new(1,0)
    
    local isDragging = false
    bg.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then isDragging = true end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then isDragging = false end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if isDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local mousePos = UserInputService:GetMouseLocation().X
            local bgPos = bg.AbsolutePosition.X; local bgSize = bg.AbsoluteSize.X
            local pct = math.clamp((mousePos - bgPos) / bgSize, 0, 1)
            fill.Size = UDim2.new(pct, 0, 1, 0)
            local val = math.floor(min + (max - min) * pct)
            label.Text = text .. ": " .. val
            callback(val)
        end
    end)
end

local isEscaping = false
local camera = workspace.CurrentCamera

-- ANA FONKSİYON
local function startEscape()
    if isEscaping then return end
    
    local char = localPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    
    if char and hrp and hum then
        isEscaping = true
        local originalCFrame = hrp.CFrame
        local destroyHeight = game.Workspace.FallenPartsDestroyHeight
        
        -- Hedef Konum ve Rotasyon (Yok olma sınırının hemen üstü)
        local escapePos = Vector3.new(0, destroyHeight + 2.5, 0)
        local escapeRot = CFrame.Angles(math.rad(90), 0, 0)
        
        -- Karakter Direncini Kapat
        hum.PlatformStand = true
        hum.AutoRotate = false
        
        -- Fiziksel Kuvvetler (BodyPosition ve BodyGyro)
        local bp = Instance.new("BodyPosition", hrp)
        bp.MaxForce = Vector3.new(1, 1, 1) * math.huge
        bp.Position = escapePos
        bp.P = 50000 
        
        local bg = Instance.new("BodyGyro", hrp)
        bg.MaxTorque = Vector3.new(1, 1, 1) * math.huge
        bg.CFrame = escapeRot
        bg.P = 500000 
        bg.D = 500    
        
        -- Anında Işınla
        hrp.CFrame = CFrame.new(escapePos) * escapeRot
        escapeBtn.Text = "Kaçılıyor (3s)..."
        
        -- 3 Saniye Bekle
        task.wait(3)
        
        -- GERİ DÖNÜŞ VE SIFIRLAMA
        if char and hrp and hum then
            bp:Destroy()
            bg:Destroy()
            
            hum.PlatformStand = false
            hum.AutoRotate = true
            
            -- Momentumu Sıfırla (Fırlamayı önler)
            hrp.AssemblyLinearVelocity = Vector3.new(0,0,0)
            hrp.AssemblyAngularVelocity = Vector3.new(0,0,0)
            
            -- Eski Yerine Işınla
            hrp.CFrame = originalCFrame
            
            -- Kamera Sabitleme
            task.wait(0.1)
            workspace.CurrentCamera.CameraType = Enum.CameraType.Custom
            workspace.CurrentCamera.CameraSubject = hum
        end
        
        escapeBtn.Text = "det kontur yememe"
        isEscaping = false
    end
end

-- 1. YÖNTEM: Buton ile çalıştırma
escapeBtn.MouseButton1Click:Connect(startEscape)

-- 2. YÖNTEM: V Tuşu ile çalıştırma (Senin tanımladığın UserInputService kullanıldı)

    if gameProcessed then return end
    
    if input.KeyCode == Enum.KeyCode.V then
        startEscape()
    end

UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    
local flySpeed = 100 -- Uçuş hızı
local flyConn


-- Değişken isimlerini değiştirdik ki eski "flyBodyVelocity" kodlarıyla karışmasın
local customFlyVel 
local customFlyOri

if input.KeyCode == Enum.KeyCode.Y and flyEnabled then
    isFlying = not isFlying
    local hrp = localPlayer.Character and localPlayer.Character:FindFirstChild("HumanoidRootPart")
    local humanoid = localPlayer.Character and localPlayer.Character:FindFirstChild("Humanoid")
    
    if hrp and humanoid then
        if isFlying then
            -- 1. ADIM: Güvenli Temizlik (Varsa bizim oluşturduğumuz eski uçma objelerini sil)
            if hrp:FindFirstChild("CustomFlyVelocity") then hrp.CustomFlyVelocity:Destroy() end
            if hrp:FindFirstChild("CustomFlyGyro") then hrp.CustomFlyGyro:Destroy() end
            if hrp:FindFirstChild("CustomFlyAttachment") then hrp.CustomFlyAttachment:Destroy() end

            -- 2. ADIM: Objeleri Özel İsimlerle Oluştur
            local att = Instance.new("Attachment", hrp)
            att.Name = "CustomFlyAttachment"

            customFlyVel = Instance.new("LinearVelocity")
            customFlyVel.Name = "CustomFlyVelocity" -- İSİM VERDİK: Başka scriptler bununla çakışmaz
            customFlyVel.MaxForce = 10^10 
            customFlyVel.VelocityConstraintMode = Enum.VelocityConstraintMode.Vector
            customFlyVel.Attachment0 = att
            customFlyVel.Parent = hrp

            customFlyOri = Instance.new("AlignOrientation")
            customFlyOri.Name = "CustomFlyGyro" -- İSİM VERDİK
            customFlyOri.MaxTorque = 10^10
            customFlyOri.Responsiveness = 200
            customFlyOri.Mode = Enum.OrientationAlignmentMode.OneAttachment
            customFlyOri.Attachment0 = att
            customFlyOri.Parent = hrp

            humanoid.PlatformStand = true

            -- 3. ADIM: Uçuş Döngüsü
            flyConn = RunService.RenderStepped:Connect(function()
                if not isFlying or not hrp.Parent then
                    if flyConn then flyConn:Disconnect() end
                    return
                end

                local camCF = workspace.CurrentCamera.CFrame
                customFlyOri.CFrame = camCF 
                
                local moveDir = Vector3.new(0, 0, 0)
                
                if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveDir = moveDir + camCF.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveDir = moveDir - camCF.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveDir = moveDir + camCF.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveDir = moveDir - camCF.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.Space) then moveDir = moveDir + Vector3.new(0, 1, 0) end
                if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then moveDir = moveDir - Vector3.new(0, 1, 0) end

                if moveDir.Magnitude > 0 then
                    -- LinearVelocity olduğu için .Velocity DEĞİL, .VectorVelocity kullanıyoruz
                    customFlyVel.VectorVelocity = moveDir.Unit * flySpeed
                else
                    customFlyVel.VectorVelocity = Vector3.new(0, 0, 0)
                end
            end)
        else
            -- 4. ADIM: Kapatma İşlemi Sırasında İsimle Bulup Yok Et
            if hrp:FindFirstChild("CustomFlyVelocity") then hrp.CustomFlyVelocity:Destroy() end
            if hrp:FindFirstChild("CustomFlyGyro") then hrp.CustomFlyGyro:Destroy() end
            if hrp:FindFirstChild("CustomFlyAttachment") then hrp.CustomFlyAttachment:Destroy() end
            if flyConn then flyConn:Disconnect() end
            
            humanoid.PlatformStand = false
            humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
        end
    end
end

 if input.KeyCode == Enum.KeyCode.R and rKeyEnabled and not isAttached then
    local targetHRP = getNearestTarget()
    if targetHRP then
        isAttached = true
        local character = localPlayer.Character
        local myHRP = character.HumanoidRootPart

                if myHumanoid then
                    myHumanoid:ChangeState(Enum.HumanoidStateType.Landed)
                end
                
                conn:Disconnect() 
                return
            end
            
            -- IŞINLANMA VE KUVVET UYGULAMA
            -- Karakteri hedefin üzerine (yatay şekilde) kilitle
            -- IŞINLANMA VE KUVVET UYGULAMA

myHRP.CFrame = CFrame.new(targetHRP.Position + Vector3.new(0, 3, 0)) * CFrame.Angles(math.rad(90), 0, 0)
            
            -- Sürekli yukarı kuvvet uygula (Bu kuvvet rakibe aktarılır)
            myHRP.AssemblyLinearVelocity = Vector3.new(0, 200
            , 0)
        end)
    end

-- Bu değişkenleri scriptin en üstüne, döngülerin dışına koy


if input.KeyCode == Enum.KeyCode.E and yonelmeEnabled then
    isYonelmeActive = not isYonelmeActive -- Durumu tersine çevir (Toggle)

    local myChar = localPlayer.Character
    local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")
    local humanoid = myChar and myChar:FindFirstChild("Humanoid")

    -- KAPATMA MODU
    if not isYonelmeActive then
        if yonelmeConn then yonelmeConn:Disconnect() end
        if currentAlignOri then currentAlignOri:Destroy() end
        if currentAtt then currentAtt:Destroy() end
        
        if humanoid then humanoid.AutoRotate = true end
        
        -- FIRLAMA ENGELLEYİCİ: Hızı ve dönme kuvvetini anında sıfırla
        if myHrp then
            myHrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            myHrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
        end
        return -- Scriptin geri kalanını çalıştırma
    end

    -- AÇMA MODU
    local targetHRP = getNearestTarget()
    if targetHRP and myHrp and humanoid then
        -- Eski bağlantılar kalmışsa temizle
        if yonelmeConn then yonelmeConn:Disconnect() end
        
        -- Force Kurulumu
        currentAtt = Instance.new("Attachment", myHrp)
        currentAlignOri = Instance.new("AlignOrientation")
        currentAlignOri.Mode = Enum.OrientationAlignmentMode.OneAttachment
        currentAlignOri.Attachment0 = currentAtt
        currentAlignOri.MaxTorque = 10^7
        currentAlignOri.Responsiveness = 200
        currentAlignOri.CFrame = CFrame.Angles(math.rad(90), 0, 0)
        currentAlignOri.Parent = myHrp

        humanoid.AutoRotate = false
        
        yonelmeConn = RunService.Heartbeat:Connect(function()
            -- Hedef veya sen yok olursan otomatik kapat
            if not targetHRP.Parent or not myHrp.Parent then
                isYonelmeActive = false
                if yonelmeConn then yonelmeConn:Disconnect() end
                if currentAlignOri then currentAlignOri:Destroy() end
                if currentAtt then currentAtt:Destroy() end
                humanoid.AutoRotate = true
                return
            end
            
            -- X ve Z'yi hedefe kitle, Y'yi (yerçekimini) serbest bırak
            local currentY = myHrp.Position.Y 
            local targetPos = targetHRP.Position
            
            myHrp.CFrame = CFrame.new(targetPos.X, currentY, targetPos.Z) * CFrame.Angles(math.rad(90), 0, 0)
        end)
    else
        -- Eğer hedef yoksa açılmasın, durumu false çek
        isYonelmeActive = false
    end
end
end)

RunService.RenderStepped:Connect(function()
    local char = localPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    
    if hum then
        if speedLoopEnabled then hum.WalkSpeed = selectedSpeed end
        if jumpLoopEnabled then hum.UseJumpPower = true; hum.JumpPower = selectedJump end
    end

    for _, p in pairs(Players:GetPlayers()) do
        if p ~= localPlayer and p.Character then
            local h = p.Character:FindFirstChild("AydoESP") or Instance.new("Highlight", p.Character)
            h.Name = "AydoESP"; h.Enabled = espEnabled
        end
    end
    
    if monsterEspEnabled then
        for _, obj in pairs(workspace:GetChildren()) do
            if obj:IsA("Model") and obj:FindFirstChildOfClass("Humanoid") and not Players:GetPlayerFromCharacter(obj) then
                local h = obj:FindFirstChild("MonsterESP") or Instance.new("Highlight", obj)
                h.Name = "MonsterESP"; h.Enabled = true; h.FillColor = Color3.new(1, 0.5, 0)
            end
        end
    end

    if isFlying and flyBodyVelocity and flyBodyGyro then
        local dir = Vector3.zero
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + camera.CFrame.RightVector end
        flyBodyVelocity.Velocity = dir.Magnitude > 0 and dir.Unit * flySpeed or Vector3.zero
        flyBodyGyro.CFrame = camera.CFrame
    end

    if noclipEnabled and char then
        for _, v in pairs(char:GetDescendants()) do if v:IsA("BasePart") then v.CanCollide = false end end
    end
    if ragdollEnabled and hum then 
            for _, t in pairs(hum:GetPlayingAnimationTracks()) do t:Stop() end
    end
end)
