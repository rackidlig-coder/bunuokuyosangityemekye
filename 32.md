local function cleanupHub()
    isDestroyed = true
    if screenGui then screenGui:Destroy() end
    for _, conn in pairs(connections) do
        if conn.Disconnect then conn:Disconnect() end
    end
    -- Highlight'ları temizle
    for _, p in pairs(Players:GetPlayers()) do
        if p.Character and p.Character:FindFirstChild("AydoESP") then p.Character.AydoESP:Destroy() end
    end
    for _, o in pairs(workspace:GetChildren()) do
        if o:FindFirstChild("MESP") then o.MESP:Destroy() end
    end
    -- Uçma objelerini temizle
    if flyBodyVelocity then flyBodyVelocity:Destroy() end
    if flyBodyGyro then flyBodyGyro:Destroy() end
end

---------------------------------------------------------
-- HEDEF BULUCU (GENEL)
---------------------------------------------------------
local function getNearestTarget()
    local myChar = localPlayer.Character
    if not myChar or not myChar:FindFirstChild("HumanoidRootPart") then return nil end
    local myPos = myChar.HumanoidRootPart.Position
    local closestHRP, minDist = nil, math.huge

    for _, player in pairs(Players:GetPlayers()) do
        if player ~= localPlayer and player.Character then
            local hrp = player.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                local dist = (hrp.Position - myPos).Magnitude
                if dist < minDist then minDist = dist; closestHRP = hrp end
            end
        end
    end

    for _, obj in pairs(workspace:GetChildren()) do
        if obj:IsA("Model") and obj ~= myChar then
            local hum = obj:FindFirstChildOfClass("Humanoid")
            local hrp = obj:FindFirstChild("HumanoidRootPart")
            if hum and hrp then
                local dist = (hrp.Position - myPos).Magnitude
                if dist < minDist then minDist = dist; closestHRP = hrp end
            end
        end
    end
    return closestHRP
end

table.insert(connections, UserInputService.InputBegan:Connect(function(input, gp)
    if isDestroyed then return end
    
    -- Menü Gizle/Aç Kısayolu (RightControl / Sağ Ctrl)
    if input.KeyCode == Enum.KeyCode.RightControl and mainFrame then
        if not loginFrame.Visible then -- Login ekranında değilse çalışsın
            mainFrame.Visible = not mainFrame.Visible
        end
    end
    
    if gp then return end

    -- FLY (G)
    if input.KeyCode == Enum.KeyCode.G and flyEnabled then
        isFlying = not isFlying
        local hrp = localPlayer.Character and localPlayer.Character:FindFirstChild("HumanoidRootPart")
        if hrp then
            if isFlying then
                flyBodyVelocity = Instance.new("BodyVelocity", hrp); flyBodyVelocity.MaxForce = Vector3.new(1,1,1)*9e9
                flyBodyGyro = Instance.new("BodyGyro", hrp); flyBodyGyro.MaxTorque = Vector3.new(1,1,1)*9e9
            else
                if flyBodyVelocity then flyBodyVelocity:Destroy() end
                if flyBodyGyro then flyBodyGyro:Destroy() end
            end
        end
    end

    -- R GRAB (Momentum Fix)
    if input.KeyCode == Enum.KeyCode.R and rKeyEnabled and not isAttached then
        local targetHRP = getNearestTarget()
        if targetHRP then
            isAttached = true
            local myChar = localPlayer.Character
            local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")
            if myHRP then
                myHRP.Anchored = true -- Fiziği durdur ve uçmayı engelle
                local conn; conn = RunService.Heartbeat:Connect(function()
                    if isDestroyed or not isAttached or not targetHRP.Parent or not myChar then 
                        isAttached = false; if conn then conn:Disconnect() end; return 
                    end
                    myHRP.CFrame = CFrame.new(targetHRP.Position + Vector3.new(0, 2, 0)) * CFrame.Angles(math.rad(90), 0, 0)
                end)
                table.insert(connections, conn)
                
                task.wait(1) 
                isAttached = false
                if conn then conn:Disconnect() end
                
                -- Havadayken fırlamayı engellemek için velocity sıfırlanıyor
                myHRP.AssemblyLinearVelocity = Vector3.zero
                myHRP.AssemblyAngularVelocity = Vector3.zero
                myHRP.Anchored = false 
            end
        end
    end

if input.KeyCode == Enum.KeyCode.T and tKeyEnabled and not isTActionActive then
    local myChar = localPlayer.Character
    if myChar and myChar:FindFirstChild("HumanoidRootPart") then
        local myHRP = myChar.HumanoidRootPart
        local targetHRP = nil
        local minDist = math.huge
        
        for _, p in pairs(Players:GetPlayers()) do
            if p ~= localPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                local dist = (p.Character.HumanoidRootPart.Position - myHRP.Position).Magnitude
                if dist < minDist then minDist = dist; targetHRP = p.Character.HumanoidRootPart end
            end
        end

        if targetHRP then
            isTActionActive = true; myHRP.Anchored = true
            -- Animasyonları Linear olarak güncelledim
            local tweenInfo1 = TweenInfo.new(0.13, Enum.EasingStyle.Linear)
            local tweenInfo2 = TweenInfo.new(0.12, Enum.EasingStyle.Linear)
            
            TweenService:Create(myHRP, tweenInfo1, {CFrame = targetHRP.CFrame * CFrame.new(0,0,-12)}):Play()
            task.wait(0.15)
            TweenService:Create(myHRP, tweenInfo2, {CFrame = targetHRP.CFrame}):Play()
            task.wait(0.15)
            
            -- Hedefe yapışık kalma döngüsü iptal edildi.
            -- Son pozisyonda sabit kalıp, yüzünü havaya (90 derece) çevirir ve 0.5 saniye bekler.
            myHRP.CFrame = CFrame.new(myHRP.Position) * CFrame.Angles(math.rad(90), 0, 0)
            task.wait(0.5)
            
            myHRP.AssemblyLinearVelocity = Vector3.zero
            myHRP.AssemblyAngularVelocity = Vector3.zero
            myHRP.Anchored = false; isTActionActive = false
        end
    end
  end
end))

table.insert(connections, RunService.RenderStepped:Connect(function()
    if isDestroyed then return end
    local char = localPlayer.Character
    if not char then return end

    -- ESP
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= localPlayer and p.Character then
            local h = p.Character:FindFirstChild("AydoESP") or Instance.new("Highlight", p.Character)
            h.Name = "AydoESP"; h.Enabled = espEnabled; h.FillColor = neonCyan; h.OutlineColor = Color3.new(1,1,1)
        end
    end
    
    if monsterEspEnabled then
        for _, o in pairs(workspace:GetChildren()) do
            if o:IsA("Model") and o:FindFirstChildOfClass("Humanoid") and not Players:GetPlayerFromCharacter(o) then
                local h = o:FindFirstChild("MESP") or Instance.new("Highlight", o)
                h.Name = "MESP"; h.Enabled = true; h.FillColor = Color3.new(1,0,0); h.OutlineColor = Color3.new(1,0,0)
            end
        end
    end

    -- FLY
    if isFlying and flyBodyVelocity and flyBodyGyro then
        local dir = Vector3.zero
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + camera.CFrame.RightVector end
        flyBodyVelocity.Velocity = dir.Magnitude > 0 and dir.Unit * flySpeed or Vector3.zero
        flyBodyGyro.CFrame = camera.CFrame
    end

    -- NOCLIP
    if noclipEnabled then
        for _,v in pairs(char:GetDescendants()) do if v:IsA("BasePart") then v.CanCollide = false end end
    end
end))
