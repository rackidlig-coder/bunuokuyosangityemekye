local function getNearestTarget()
    local myChar = localPlayer.Character
    if not myChar or not myChar:FindFirstChild("HumanoidRootPart") then return nil end
    local myPos = myChar.HumanoidRootPart.Position
    local closestHRP, minDist = nil, math.huge

    for _, player in pairs(Players:GetPlayers()) do
        if player ~= localPlayer and player.Character then
            local hrp = player.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                local dist = (hrp.Position - myPos).Magnitude
                if dist < minDist then minDist = dist; closestHRP = hrp end
            end
        end
    end

    for _, obj in pairs(workspace:GetChildren()) do
        if obj:IsA("Model") and obj ~= myChar then
            local hum = obj:FindFirstChildOfClass("Humanoid")
            local hrp = obj:FindFirstChild("HumanoidRootPart")
            if hum and hrp then
                local dist = (hrp.Position - myPos).Magnitude
                if dist < minDist then minDist = dist; closestHRP = hrp end
            end
        end
    end
    return closestHRP
end

local function createSlider(text, parent, min, max, default, callback)
    local container = Instance.new("Frame", parent)
    container.Size = UDim2.new(0.9, 0, 0, 40); container.BackgroundTransparency = 1
    
    local label = Instance.new("TextLabel", container)
    label.Size = UDim2.new(1, 0, 0, 15); label.BackgroundTransparency = 1
    label.Text = text .. ": " .. default; label.TextColor3 = Color3.new(1,1,1)
    label.Font = Enum.Font.GothamBold; label.TextSize = 12; label.TextXAlignment = Enum.TextXAlignment.Left
    
    local bg = Instance.new("Frame", container)
    bg.Size = UDim2.new(1, 0, 0, 10); bg.Position = UDim2.new(0, 0, 0, 20)
    bg.BackgroundColor3 = Color3.fromRGB(30, 30, 35)
    Instance.new("UICorner", bg).CornerRadius = UDim.new(1,0)
    local stroke = Instance.new("UIStroke", bg); stroke.Color = Color3.fromRGB(60, 60, 70)
    
    local fill = Instance.new("Frame", bg)
    fill.Size = UDim2.new((default-min)/(max-min), 0, 1, 0)
    fill.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
    Instance.new("UICorner", fill).CornerRadius = UDim.new(1,0)
    
    local isDragging = false
    bg.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then isDragging = true end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then isDragging = false end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if isDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local mousePos = UserInputService:GetMouseLocation().X
            local bgPos = bg.AbsolutePosition.X; local bgSize = bg.AbsoluteSize.X
            local pct = math.clamp((mousePos - bgPos) / bgSize, 0, 1)
            fill.Size = UDim2.new(pct, 0, 1, 0)
            local val = math.floor(min + (max - min) * pct)
            label.Text = text .. ": " .. val
            callback(val)
        end
    end)
end

local isEscaping = false
local camera = workspace.CurrentCamera

escapeBtn.MouseButton1Click:Connect(function()
    if isEscaping then return end
    local char = localPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    
    if char and hrp and hum then
        isEscaping = true
        local originalCFrame = hrp.CFrame
        local destroyHeight = game.Workspace.FallenPartsDestroyHeight
        
        -- Hedef Konum (Yok olma çizgisinin 2.5 stud üstü)
        local escapePos = Vector3.new(0, destroyHeight + 2.5, 0)
        local escapeRot = CFrame.Angles(math.rad(90), 0, 0)
        
        -- Karakterin direnmesini engelle
        hum.PlatformStand = true
        hum.AutoRotate = false
        
        -- GÜÇLÜ GÖVDE POZİSYONU (BodyPosition)
        local bp = Instance.new("BodyPosition", hrp)
        bp.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
        bp.Position = escapePos
        bp.P = 50000 -- Hedefe anında kitlenmesi için yüksek güç
        
        -- EKSTRA GÜÇLÜ DÖNÜŞ (BodyGyro)
        local bg = Instance.new("BodyGyro", hrp)
        bg.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
        bg.CFrame = escapeRot
        bg.P = 500000 -- 90 dereceyi beton gibi tutması için devasa güç
        bg.D = 500    -- Titremeyi tamamen yok eden direnç
        
        -- Anında ışınla (Fizik motoru gelene kadar beklemesin)
        hrp.CFrame = CFrame.new(escapePos) * escapeRot
        escapeBtn.Text = "Kaçılıyor (3s)..."
        
        -- 3 Saniye Bekle
        task.wait(3)
        
        -- GERİ DÖNÜŞ VE SIFIRLAMA
        if char and hrp and hum then
            bp:Destroy()
            bg:Destroy()
            
            hum.PlatformStand = false
            hum.AutoRotate = true
            
            hrp.Velocity = Vector3.new(0,0,0) -- Hızı sıfırla ki fırlamasın
            hrp.CFrame = originalCFrame
            
            -- Kamera Reset
            task.wait(0.1)
            camera.CameraType = Enum.CameraType.Custom
            camera.CameraSubject = hum
            camera.Focus = hrp.CFrame
        end
        
        escapeBtn.Text = "Kaçma Özelliği"
        isEscaping = false
    end
end)

UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    
    if input.KeyCode == Enum.KeyCode.G and flyEnabled then
        isFlying = not isFlying
        local hrp = localPlayer.Character and localPlayer.Character:FindFirstChild("HumanoidRootPart")
        if hrp then
            if isFlying then
                flyBodyVelocity = Instance.new("BodyVelocity", hrp); flyBodyVelocity.MaxForce = Vector3.new(1,1,1)*9e9
                flyBodyGyro = Instance.new("BodyGyro", hrp); flyBodyGyro.MaxTorque = Vector3.new(1,1,1)*9e9
            else
                if flyBodyVelocity then flyBodyVelocity:Destroy() end
                if flyBodyGyro then flyBodyGyro:Destroy() end
            end
        end
    end

    if input.KeyCode == Enum.KeyCode.R and rKeyEnabled and not isAttached then
        local targetHRP = getNearestTarget()
        if targetHRP then
            isAttached = true
            local myHRP = localPlayer.Character.HumanoidRootPart
            local startTime = tick()
            local conn
            conn = RunService.Heartbeat:Connect(function()
                if tick() - startTime > 0.5 or not targetHRP.Parent or not myHRP.Parent then
                    isAttached = false; conn:Disconnect(); return
                end
                myHRP.CFrame = CFrame.new(targetHRP.Position + Vector3.new(0, 2, 0)) * CFrame.Angles(math.rad(90), 0, 0)
                myHRP.AssemblyLinearVelocity = Vector3.zero
            end)
        end
    end
end)

RunService.RenderStepped:Connect(function()
    local char = localPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    
    if hum then
        if speedLoopEnabled then hum.WalkSpeed = selectedSpeed end
        if jumpLoopEnabled then hum.UseJumpPower = true; hum.JumpPower = selectedJump end
    end

    for _, p in pairs(Players:GetPlayers()) do
        if p ~= localPlayer and p.Character then
            local h = p.Character:FindFirstChild("AydoESP") or Instance.new("Highlight", p.Character)
            h.Name = "AydoESP"; h.Enabled = espEnabled
        end
    end
    
    if monsterEspEnabled then
        for _, obj in pairs(workspace:GetChildren()) do
            if obj:IsA("Model") and obj:FindFirstChildOfClass("Humanoid") and not Players:GetPlayerFromCharacter(obj) then
                local h = obj:FindFirstChild("MonsterESP") or Instance.new("Highlight", obj)
                h.Name = "MonsterESP"; h.Enabled = true; h.FillColor = Color3.new(1, 0.5, 0)
            end
        end
    end

    if isFlying and flyBodyVelocity and flyBodyGyro then
        local dir = Vector3.zero
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + camera.CFrame.RightVector end
        flyBodyVelocity.Velocity = dir.Magnitude > 0 and dir.Unit * flySpeed or Vector3.zero
        flyBodyGyro.CFrame = camera.CFrame
    end

    if noclipEnabled and char then
        for _, v in pairs(char:GetDescendants()) do if v:IsA("BasePart") then v.CanCollide = false end end
    end
    if ragdollEnabled and hum then 
            for _, t in pairs(hum:GetPlayingAnimationTracks()) do t:Stop() end
    end
end)
